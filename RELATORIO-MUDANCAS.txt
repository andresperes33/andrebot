**JS de compartilhar** (em `bot/templates/bot/base.html`):

```js
document.addEventListener('click', function (e) {
    var btn = e.target.closest('.promo-card-share');
    if (!btn) return;
    e.preventDefault();
    e.stopPropagation();
    var url = btn.getAttribute('data-share-url');
    var titulo = btn.getAttribute('data-share-title') || 'Nitro Tech';
    if (navigator.share) {
        navigator.share({ title: titulo, url: url }).catch(function () {});
    } else if (navigator.clipboard) {
        navigator.clipboard.writeText(url).then(function () {
            var toast = document.getElementById('toast');
            if (toast) {
                toast.textContent = 'Link copiado!';
                toast.classList.add('show');
                setTimeout(function () { toast.classList.remove('show'); }, 2000);
            }
        });
    }
});
```

```js
(function () {
    var banner = document.getElementById('channelsBanner');
    if (!banner) return;
    var lastY = window.scrollY;
    var limite = 60;
    window.addEventListener('scroll', function () {
        var y = window.scrollY;
        if (y > limite && y > lastY) {
            banner.classList.add('hidden');
        } else if (y <= limite || y < lastY) {
            banner.classList.remove('hidden');
        }
        lastY = y;
    }, { passive: true });
})();
```

**JS (início de `bot/templates/bot/promos.html`):**

```js
(function () {
    var track = document.getElementById('carrosselTrack');
    if (!track) return;
    var slides = track.children.length;
    if (slides < 2) return;
    var dotsWrap = document.getElementById('carrosselDots');
    var prev = document.querySelector('.carrossel-prev');
    var next = document.querySelector('.carrossel-next');
    var idx = 0;
    var timer = null;

    function updateDots() {
        var dots = dotsWrap.children;
        for (var i = 0; i < dots.length; i++) {
            dots[i].classList.toggle('active', i === idx);
        }
    }
    function go(n) {
        idx = ((n % slides) + slides) % slides;
        var step = track.children[0].offsetWidth;
        track.style.transform = 'translateX(' + (-idx * step) + 'px)';
        updateDots();
    }
    function restart() {
        if (timer) clearInterval(timer);
        timer = setInterval(function () { go(idx + 1); }, 8000);
    }
    for (var i = 0; i < slides; i++) {
        (function (i) {
            var d = document.createElement('button');
            d.className = 'carrossel-dot' + (i === 0 ? ' active' : '');
            d.setAttribute('aria-label', 'Ir para o banner ' + (i + 1));
            d.addEventListener('click', function () { go(i); restart(); });
            dotsWrap.appendChild(d);
        })(i);
    }
    prev.addEventListener('click', function () { go(idx - 1); restart(); });
    next.addEventListener('click', function () { go(idx + 1); restart(); });
    restart();
    var carrossel = document.getElementById('carrossel');
    carrossel.addEventListener('mouseenter', function () { if (timer) clearInterval(timer); });
    carrossel.addEventListener('mouseleave', restart);

    // Swipe no mobile
    var startX = null;
    track.addEventListener('touchstart', function (e) { startX = e.touches[0].clientX; }, { passive: true });
    track.addEventListener('touchend', function (e) {
        if (startX === null) return;
        var delta = e.changedTouches[0].clientX - startX;
        if (delta < -40) { go(idx + 1); restart(); }
        else if (delta > 40) { go(idx - 1); restart(); }
        startX = null;
    }, { passive: true });

    window.addEventListener('resize', function () {
        var step = track.children[0].offsetWidth;
        track.style.transform = 'translateX(' + (-idx * step) + 'px)';
    });
})();
```

```python
def site_base_url(request):
    """URL base do site, sempre em HTTPS (mesmo atrás de proxy/Cloudflare)."""
    url = request.build_absolute_uri('/')
    if url.startswith('http://'):
        url = 'https://' + url[len('http://'):]
    return url.rstrip('/')
```

## 14. CLASSIFICAÇÃO DE CONSOLE (Nintendo Switch não vira TV)

**`bot/classifier.py`** — bloco antes da regra de TV:

```python
# CONSOLE tem prioridade sobre 'oled'/'qled' da TV: 'Nintendo Switch
# Console OLED 64gb' é um CONSOLE, não uma TV, mesmo com 'OLED' no título.
for alvo in (titulo, texto,):
    if not alvo:
        continue
    alvo_norm = _norm(_limpar_compat(alvo))
    if not alvo_norm:
        continue
    if re.search(r'\b(?:headset|headphone|fone\b|fones\b|auricular|earbuds?)\b', alvo_norm):
        continue  # fone/headset para console não é console
    eh_console = re.search(
        r'(?:\bnintendo\s*switch\b'
        r'|\bnintendo\b(?!\s*(?:para|compat[ií]vel|fone|controle|acess[oó]rio))'
        r'|\bswitch\b(?=\s*(?:nintendo|oled|lite|2\b|\d|joy|v2))'
        r'|\bplaystation\b(?!\s*(?:para|compat[ií]vel|fone|headset|controle|acess[oó]rio|computador|pc\b))'
        r'|\bxbox\b(?!\s*(?:para|fone|controle|headset|acess[oó]rio|series\s*[sx]\s*compat))'
        r'|\bsteam\s*deck\b|\bok\s*1\b|\banbernic\b|\bhandheld\b|\bconsole\b'
        r'|\bps[0-9]\b(?!ps[0-9]|,[^.\n]*ps[0-9]|[^.\n]*\b(?:para|compat[ií]vel|fone|headset|controle|ssd|disco|jogo|acess[oó]rio)\b))',
        alvo_norm,
    )
    if eh_console:
        return 'console'
```
